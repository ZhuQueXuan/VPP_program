"""Read-only source/artifact audit; write only to an explicitly new work directory."""
import argparse
from pathlib import Path
from core_io import BASE,ROOT,read,write,sha

def verify_to(output):
    target=Path(output).resolve();work=(ROOT/'work').resolve()
    try:target.relative_to(work)
    except ValueError:raise ValueError('audit destination must be a new directory under project/work')
    if target.exists():raise FileExistsError('refuse to overwrite an existing directory or receipt')
    f=read(BASE/'source_freeze.json');snapshot=read(BASE/'final_snapshot.json');rows=[]
    for group,root in [('source',BASE),('protected',ROOT)]:
        for name,expected in f[group].items():
            actual=sha(root/name);rows.append(dict(group=group,path=str(root/name),match=actual==expected))
    for name,expected in snapshot['files'].items():rows.append(dict(group='current_artifact',path=str(ROOT/name),match=sha(ROOT/name)==expected))
    if not all(v['match'] for v in rows):raise RuntimeError('identity mismatch; no receipt written')
    target.mkdir(parents=True,exist_ok=False);write(target/'verification.json',dict(status='passed',files_checked=len(rows),original_receipts_not_modified=True,rows=rows,real_validation=False,scope='identity/read-only audit; algorithm numerical verification remains independent_checks.json'))
    return dict(status='passed',files_checked=len(rows),output=str(target/'verification.json'))

if __name__=='__main__':
    parser=argparse.ArgumentParser(description=__doc__);parser.add_argument('--output',required=True,help='a directory that does not exist under project/work');args=parser.parse_args();print(verify_to(args.output))
